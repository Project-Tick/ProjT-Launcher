set(LAUNCHER_TESTS "")

function(launcher_add_test source name)
    ecm_add_test(${source} LINK_LIBRARIES Launcher_logic Qt${QT_VERSION_MAJOR}::Test
        TEST_NAME ${name})
    list(APPEND LAUNCHER_TESTS ${name})
    set(LAUNCHER_TESTS "${LAUNCHER_TESTS}" PARENT_SCOPE)
endfunction()
